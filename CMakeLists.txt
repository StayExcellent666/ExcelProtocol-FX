cmake_minimum_required(VERSION 3.28...3.31)
# Bootstrap and metadata originate from the official OBS plugin template.
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version} LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ExcelProtocol FX currently supports Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FX_VERSION "${PROJECT_VERSION}" CACHE STRING "Release version including optional -beta.N")
set(FX_GITHUB_REPOSITORY "StayExcellent666/ExcelProtocol-FX" CACHE STRING "Official GitHub owner/repository")
if(NOT FX_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(-beta\\.[0-9]+)?$")
  message(FATAL_ERROR "FX_VERSION must be major.minor.patch or major.minor.patch-beta.N")
endif()
if(NOT FX_GITHUB_REPOSITORY MATCHES "^[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+$")
  message(FATAL_ERROR "FX_GITHUB_REPOSITORY must be owner/repository")
endif()
set(FX_DEPS "${CMAKE_SOURCE_DIR}/.deps" CACHE PATH "Prepared dependencies")
list(PREPEND CMAKE_PREFIX_PATH "${FX_DEPS}/qt6")
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets Network)
set(FX_QT_TLS_BACKEND "${FX_DEPS}/qt6/plugins/tls/qschannelbackend.dll")
if(NOT EXISTS "${FX_QT_TLS_BACKEND}")
  message(FATAL_ERROR "The Qt Schannel TLS backend is missing; run scripts/prepare-deps.ps1")
endif()
foreach(name IN ITEMS obs obs-frontend-api)
  if(NOT EXISTS "${FX_DEPS}/obs-sdk/${name}.lib")
    message(FATAL_ERROR "Run scripts/prepare-deps.ps1 in an x64 MSVC developer shell first")
  endif()
  add_library(${name} SHARED IMPORTED GLOBAL)
  set_target_properties(${name} PROPERTIES
    IMPORTED_IMPLIB "${FX_DEPS}/obs-sdk/${name}.lib"
    IMPORTED_LOCATION "${FX_DEPS}/obs-runtime/bin/64bit/${name}.dll"
    INTERFACE_INCLUDE_DIRECTORIES "${FX_DEPS}/obs-studio/libobs;${FX_DEPS}/obs-studio/frontend/api;${FX_DEPS}/obs-sdk")
endforeach()
add_library(excelprotocol-fx MODULE src/plugin-main.cpp src/effects.cpp src/raid-battle.cpp src/raid-audio.cpp src/dock.cpp src/remote-client.cpp src/update-controller.cpp src/update-policy.cpp)
target_link_libraries(excelprotocol-fx PRIVATE obs obs-frontend-api Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network user32 advapi32 winmm)
target_compile_definitions(excelprotocol-fx PRIVATE FX_VERSION="${FX_VERSION}" FX_GITHUB_REPOSITORY="${FX_GITHUB_REPOSITORY}" NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(excelprotocol-fx PRIVATE /utf-8 /W4 /permissive- /EHsc)
set_target_properties(excelprotocol-fx PROPERTIES PREFIX "")
add_executable(excelprotocol-fx-updater WIN32 updater/main.cpp)
target_link_libraries(excelprotocol-fx-updater PRIVATE bcrypt shell32)
target_compile_definitions(excelprotocol-fx-updater PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(excelprotocol-fx-updater PRIVATE /utf-8 /W4 /EHsc)
# The helper must remain usable after OBS closes, without Qt or a redist dependency.
set_property(TARGET excelprotocol-fx-updater PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
install(TARGETS excelprotocol-fx LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
install(TARGETS excelprotocol-fx-updater RUNTIME DESTINATION data/obs-plugins/excelprotocol-fx)
install(FILES "${FX_QT_TLS_BACKEND}" DESTINATION data/obs-plugins/excelprotocol-fx/tls)
install(DIRECTORY data/ DESTINATION data/obs-plugins/excelprotocol-fx)
install(FILES LICENSE DESTINATION data/obs-plugins/excelprotocol-fx)
include(CTest)
if(BUILD_TESTING)
  add_executable(update-policy-test tests/update-policy-test.cpp src/update-policy.cpp)
  target_link_libraries(update-policy-test PRIVATE Qt6::Core)
  target_include_directories(update-policy-test PRIVATE src)
  target_compile_options(update-policy-test PRIVATE /utf-8 /EHsc)
  add_test(NAME update-policy COMMAND update-policy-test)

  target_link_libraries(effects-test PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets obs user32 winmm)
  target_include_directories(effects-test PRIVATE src)
  target_compile_definitions(effects-test PRIVATE FX_TESTING NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(effects-test PRIVATE /utf-8 /EHsc)
  add_test(NAME effects COMMAND effects-test)
  add_executable(raid-battle-test tests/raid-battle-test.cpp src/raid-battle.cpp)
  target_link_libraries(raid-battle-test PRIVATE Qt6::Core Qt6::Gui)
  target_include_directories(raid-battle-test PRIVATE src)
  target_compile_options(raid-battle-test PRIVATE /utf-8 /EHsc)
  add_test(NAME raid-battle COMMAND raid-battle-test)
  add_executable(raid-audio-test tests/raid-audio-test.cpp src/raid-audio.cpp)
  target_link_libraries(raid-audio-test PRIVATE Qt6::Core winmm)
  target_include_directories(raid-audio-test PRIVATE src)

  target_compile_options(raid-audio-test PRIVATE /utf-8 /EHsc)

endif()
